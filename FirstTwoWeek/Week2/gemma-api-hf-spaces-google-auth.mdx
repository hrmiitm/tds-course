# Lab 2.1 — Gemma Model API on HuggingFace Spaces + Google Auth

**What you'll build:** A live, public FastAPI service on HuggingFace Spaces that:
- Accepts text generation requests at `POST /generate`
- Is protected by Google OAuth — only authenticated users can call it
- Uses Gemma 3 (via Ollama locally, or HuggingFace Inference API on Spaces)
- Returns JWT tokens for subsequent API calls

**Live URL after completion:** `https://your-username-tds-lab-2.hf.space`

---

## Architecture

```
Client
  │
  ├─→ GET /auth/login  ──────────→ Google OAuth
  │                                     │
  │   ←── redirect with code ───────────┘
  │
  ├─→ GET /auth/callback?code=...
  │       └─→ exchange code → access token → user info → JWT
  │       ←── {"access_token": "eyJ..."}
  │
  └─→ POST /generate  (Authorization: Bearer eyJ...)
          └─→ validate JWT → call Gemma → return text
```

<summary>**Step 3 — Write the Config and Auth Modules**</summary>

```python title="auth.py"
from fastapi import HTTPException, Request
from jose import JWTError, jwt
from datetime import datetime, timedelta
from pydantic_settings import BaseSettings, SettingsConfigDict
import httpx

class Settings(BaseSettings):
    google_client_id: str
    google_client_secret: str
    secret_key: str
    redirect_uri: str = "http://localhost:8000/auth/callback"
    allowed_domain: str = ""          # e.g. "ds.study.iitm.ac.in"
    jwt_expire_hours: int = 24

    model_config = SettingsConfigDict(env_file=".env", case_sensitive=False)

settings = Settings()

GOOGLE_AUTH_URL = "https://accounts.google.com/o/oauth2/v2/auth"
GOOGLE_TOKEN_URL = "https://oauth2.googleapis.com/token"
GOOGLE_USERINFO_URL = "https://www.googleapis.com/oauth2/v2/userinfo"
ALGORITHM = "HS256"


# ── JWT helpers ────────────────────────────────────────────────

def create_jwt(data: dict) -> str:
    payload = {**data, "exp": datetime.utcnow() + timedelta(hours=settings.jwt_expire_hours)}
    return jwt.encode(payload, settings.secret_key, algorithm=ALGORITHM)


def decode_jwt(token: str) -> dict:
    try:
        return jwt.decode(token, settings.secret_key, algorithms=[ALGORITHM])
    except JWTError as e:
        raise HTTPException(status_code=401, detail=f"Invalid token: {e}")


# ── Dependency ─────────────────────────────────────────────────

def get_current_user(request: Request) -> dict:
    auth = request.headers.get("Authorization", "")
    if not auth.startswith("Bearer "):
        raise HTTPException(401, "Missing Bearer token. Visit /auth/login first.")
    return decode_jwt(auth.split(" ", 1)[1])


# ── OAuth helpers ──────────────────────────────────────────────

def build_google_auth_url() -> str:
    params = {
        "client_id": settings.google_client_id,
        "redirect_uri": settings.redirect_uri,
        "response_type": "code",
        "scope": "openid email profile",
        "access_type": "offline",
        "prompt": "select_account",
    }
    query = "&".join(f"{k}={v}" for k, v in params.items())
    return f"{GOOGLE_AUTH_URL}?{query}"


async def exchange_code_for_user(code: str) -> dict:
    async with httpx.AsyncClient() as client:
        # Exchange authorization code for tokens
        token_resp = await client.post(GOOGLE_TOKEN_URL, data={
            "client_id": settings.google_client_id,
            "client_secret": settings.google_client_secret,
            "code": code,
            "redirect_uri": settings.redirect_uri,
            "grant_type": "authorization_code",
        })
        tokens = token_resp.json()
        if "error" in tokens:
            raise HTTPException(400, f"Google token error: {tokens['error_description']}")

        # Get user profile
        user_resp = await client.get(
            GOOGLE_USERINFO_URL,
            headers={"Authorization": f"Bearer {tokens['access_token']}"},
        )
        return user_resp.json()


def check_allowed_domain(email: str):
    if settings.allowed_domain and not email.endswith(f"@{settings.allowed_domain}"):
        raise HTTPException(
            403,
            f"Access restricted to @{settings.allowed_domain} accounts. "
            f"Your email ({email}) is not allowed."
        )
```

Create the `.env` file:

```bash title=".env"
GOOGLE_CLIENT_ID=your-client-id.apps.googleusercontent.com
GOOGLE_CLIENT_SECRET=your-client-secret
SECRET_KEY=generate-a-32-char-random-key-here

ALLOWED_DOMAIN=
```


<summary>**Step 4 — Write the LLM Module**</summary>

This module tries Ollama locally first, falls back to HuggingFace Inference API (works on Spaces):

```python title="llm.py"
import httpx
import os
from pydantic import BaseModel

class GenerateRequest(BaseModel):
    prompt: str
    max_tokens: int = 256
    temperature: float = 0.7

class GenerateResponse(BaseModel):
    text: str
    model: str
    tokens_used: int | None = None


# ── Local Ollama ───────────────────────────────────────────────

async def generate_ollama(req: GenerateRequest) -> GenerateResponse:
    async with httpx.AsyncClient(timeout=60.0) as client:
        resp = await client.post(
            "http://localhost:11434/api/generate",
            json={
                "model": "gemma3",
                "prompt": req.prompt,
                "stream": False,
                "options": {
                    "num_predict": req.max_tokens,
                    "temperature": req.temperature,
                },
            },
        )
        resp.raise_for_status()
        data = resp.json()
        return GenerateResponse(
            text=data["response"],
            model="gemma3-local",
            tokens_used=data.get("eval_count"),
        )


# ── HuggingFace Inference API (for Spaces deployment) ─────────

HF_MODEL = "google/gemma-3-1b-it"   # free, fast
HF_API_KEY = os.getenv("HF_API_KEY", "")

async def generate_hf_api(req: GenerateRequest) -> GenerateResponse:
    headers = {"Authorization": f"Bearer {HF_API_KEY}"} if HF_API_KEY else {}
    async with httpx.AsyncClient(timeout=60.0) as client:
        resp = await client.post(
            f"https://api-inference.huggingface.co/models/{HF_MODEL}",
            headers=headers,
            json={
                "inputs": req.prompt,
                "parameters": {
                    "max_new_tokens": req.max_tokens,
                    "temperature": req.temperature,
                    "return_full_text": False,
                },
            },
        )
        resp.raise_for_status()
        data = resp.json()
        # HF returns a list
        text = data[0]["generated_text"] if isinstance(data, list) else data.get("generated_text", "")
        return GenerateResponse(text=text, model=HF_MODEL)


# ── Auto-detect which backend to use ──────────────────────────

async def generate(req: GenerateRequest) -> GenerateResponse:
    """Try Ollama first (local dev), fall back to HF API (production)"""
    use_hf = os.getenv("USE_HF_API", "false").lower() == "true"

    if not use_hf:
        try:
            return await generate_ollama(req)
        except (httpx.ConnectError, httpx.TimeoutException):
            # Ollama not running — fall back to HF
            pass

    return await generate_hf_api(req)
```

<summary>**Step 5 — Write the Main FastAPI App**</summary>

```python title="main.py"
from fastapi import FastAPI, Depends, HTTPException
from fastapi.responses import RedirectResponse, HTMLResponse
from auth import (
    settings, get_current_user, create_jwt,
    build_google_auth_url, exchange_code_for_user, check_allowed_domain,
)
from llm import GenerateRequest, GenerateResponse, generate
import structlog

log = structlog.get_logger()

app = FastAPI(
    title="TDS Lab 2 — Gemma API",
    description="Text generation API powered by Gemma, protected by Google OAuth",
    version="1.0.0",
)


# ── Auth Routes ────────────────────────────────────────────────

@app.get("/", response_class=HTMLResponse, include_in_schema=False)
def root():
    return """
    <html><body style="font-family: sans-serif; padding: 2rem;">
      <h1>🤖 TDS Lab 2 — Gemma API</h1>
      <p>A text generation API protected by Google OAuth.</p>
      <ul>
        <li><a href="/auth/login">Login with Google</a></li>
        <li><a href="/docs">API Docs (Swagger)</a></li>
        <li><a href="/health">Health Check</a></li>
      </ul>
    </body></html>
    """

@app.get("/health")
def health():
    return {"status": "ok", "version": "1.0.0"}

@app.get("/auth/login", summary="Redirect to Google OAuth login")
def login():
    """Start the OAuth flow. Opens Google login page."""
    return RedirectResponse(build_google_auth_url())

@app.get("/auth/callback", summary="Google OAuth callback")
async def callback(code: str):
    """
    Google redirects here after login.
    Exchanges the code for a JWT you can use on /generate.
    """
    user_info = await exchange_code_for_user(code)
    email = user_info.get("email", "")

    # Optional: restrict to specific domain
    check_allowed_domain(email)

    log.info("User logged in", email=email)

    token = create_jwt({
        "sub": user_info["id"],
        "email": email,
        "name": user_info.get("name", ""),
        "picture": user_info.get("picture", ""),
    })

    return {
        "access_token": token,
        "token_type": "bearer",
        "user": {"email": email, "name": user_info.get("name", "")},
        "usage": "Add to requests: Authorization: Bearer <access_token>",
    }

@app.get("/me", summary="Get current user info")
def me(user: dict = Depends(get_current_user)):
    """Returns info about the authenticated user."""
    return {"email": user["email"], "name": user["name"]}


# ── Generation Route ───────────────────────────────────────────

@app.post(
    "/generate",
    response_model=GenerateResponse,
    summary="Generate text with Gemma",
)
async def generate_text(
    req: GenerateRequest,
    user: dict = Depends(get_current_user),   # ← requires auth
):
    """
    Generate text using Gemma.
    Requires a valid JWT from /auth/callback in the Authorization header.
    """
    log.info("Generation request", email=user["email"], prompt_len=len(req.prompt))

    try:
        result = await generate(req)
        log.info("Generation complete", model=result.model, tokens=result.tokens_used)
        return result
    except Exception as e:
        log.error("Generation failed", error=str(e))
        raise HTTPException(500, f"Generation failed: {str(e)}")
```

<summary>**Step 7 — Write the Dockerfile**</summary>

```dockerfile title="Dockerfile"
# ── Builder Stage ──────────────────────────────────────────────
FROM python:3.12-slim AS builder

WORKDIR /app

RUN pip install uv

COPY pyproject.toml uv.lock* ./
RUN uv sync --frozen --no-dev

# ── Runtime Stage ──────────────────────────────────────────────
FROM python:3.12-slim AS runtime

# HuggingFace Spaces requires UID 1000
RUN useradd -m -u 1000 user
USER user
WORKDIR /home/user/app

# Copy installed packages from builder
COPY --from=builder --chown=user /app/.venv /home/user/app/.venv
ENV PATH="/home/user/app/.venv/bin:$PATH"

# Copy application code
COPY --chown=user . .

# HuggingFace Spaces uses port 7860
EXPOSE 7860

# Use HF Inference API in production (not local Ollama)
ENV USE_HF_API=true

CMD ["uvicorn", "main:app", "--host", "0.0.0.0", "--port", "7860"]
```

Create `.dockerignore`:
```text title=".dockerignore"
.env
.env.*
__pycache__
*.pyc
.git
.gitignore
.venv
*.egg-info
```

Build and test locally:
```bash
# Build the image
docker build -t tds-lab-2 .

# Run with env vars (using your .env file)
docker run -p 7860:7860 \
  --env-file .env \
  -e USE_HF_API=false \
  tds-lab-2

# Test
curl http://localhost:7860/health
```

## Troubleshooting

| Problem | Fix |
|---------|-----|
| `redirect_uri_mismatch` | Add the exact Space URL to Google Console authorized URIs |
| `401 Unauthorized` on /generate | Token expired (24h) — re-login at `/auth/login` |
| Space stuck "Building" | Check Logs tab — usually a Python import error |
| HF model returns `503` | Model loading, wait 20s and retry |
| Ollama not found locally | Run `ollama pull gemma3` first |
